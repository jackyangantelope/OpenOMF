cmake_minimum_required(VERSION 3.20)
set(PICO_BOARD adafruit_fruit_jam CACHE STRING "Public target" FORCE)
set(PICO_PSRAM_CS_PIN 47 CACHE STRING "PSRAM CS" FORCE)
set(PICO_PSRAM_SIZE_BYTES 0x800000 CACHE STRING "PSRAM size" FORCE)
set(PICO_STACK_SIZE 32768)
set(PICO_CORE1_STACK_SIZE 4096)
if(NOT DEFINED PICO_SDK_PATH OR NOT DEFINED PICO_PIO_USB_PATH)
  message(FATAL_ERROR "Set PICO_SDK_PATH and PICO_PIO_USB_PATH")
endif()
set(PORT_ROOT "${CMAKE_CURRENT_LIST_DIR}")
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(picoOpenOmfNative C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
set(LIBS "${PORT_ROOT}/deps/libs")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(BUILD_STATIC ON CACHE BOOL "" FORCE)
set(SKIP_INSTALL_ALL ON)
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
add_compile_options(-O2 -ffunction-sections -fdata-sections
  "-ffile-prefix-map=${PORT_ROOT}=."
  "-ffile-prefix-map=${PICO_SDK_PATH}=pico-sdk"
  "-ffile-prefix-map=${PICO_PIO_USB_PATH}=Pico-PIO-USB")
set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${LIBS}/zlib" zlib EXCLUDE_FROM_ALL)
set(ZLIB_LIBRARY zlibstatic CACHE STRING "" FORCE)
set(ZLIB_INCLUDE_DIR "${LIBS}/zlib" CACHE PATH "" FORCE)
target_include_directories(zlibstatic PUBLIC "${LIBS}/zlib" "${CMAKE_BINARY_DIR}/zlib")
add_library(ZLIB::ZLIB ALIAS zlibstatic)
set(PNG_SHARED OFF CACHE BOOL "" FORCE)
set(PNG_TESTS OFF CACHE BOOL "" FORCE)
set(PNG_TOOLS OFF CACHE BOOL "" FORCE)
set(PNG_ARM_NEON off CACHE STRING "" FORCE)
set(PNG_HARDWARE_OPTIMIZATIONS OFF CACHE BOOL "" FORCE)
set(PNG_BUILD_ZLIB ON CACHE BOOL "" FORCE)
add_subdirectory("${LIBS}/libpng" png EXCLUDE_FROM_ALL)
target_link_libraries(png_static PUBLIC zlibstatic)
target_include_directories(png_static PUBLIC "${CMAKE_BINARY_DIR}/png")
set(LIBXMP_DISABLE_DEPACKERS ON CACHE BOOL "" FORCE)
set(LIBXMP_DISABLE_PROWIZARD ON CACHE BOOL "" FORCE)
add_subdirectory("${LIBS}/libxmp" xmp EXCLUDE_FROM_ALL)
add_library(confuse_native STATIC "${LIBS}/libconfuse/src/confuse.c"
  "${LIBS}/libconfuse/src/reallocarray.c" "platform/generated/confuse_lexer.c")
target_include_directories(confuse_native PUBLIC "${LIBS}/libconfuse/src")
target_compile_definitions(confuse_native PRIVATE CONFUSE_BAREMETAL=1 PACKAGE_VERSION="3.3" PACKAGE_STRING="libconfuse-3.3"
  HAVE_UNISTD_H=1 HAVE_SYS_STAT_H=1 HAVE_STRING_H=1 HAVE_STRINGS_H=1 HAVE_STRDUP=1 HAVE_STRNDUP=1 HAVE_STRCASECMP=1 HAVE_FMEMOPEN=1)
include("platform/native/omf_sources.cmake")
target_compile_definitions(omf_core PRIVATE OMF_PICO=1 MINIZ_NO_TIME=1)
target_include_directories(omf_core PRIVATE
  "${PORT_ROOT}/deps/native-headers/SDL2"
  "platform/native/compat")
target_link_libraries(omf_core PUBLIC png_static xmp_static confuse_native pico_stdlib)
set(JACK_WAVE1_HDMI_AUDIO ON)
set(JACK_WAVE1_CONTINUOUS_SILENCE ON)
add_subdirectory("platform/runtime" runtime)
target_compile_definitions(jack_wave1_runtime PRIVATE JACK_WAVE1_USB_NONBLOCKING=1 JACK_WAVE1_MAX_FILES=1)
target_compile_definitions(jack_wave1_runtime PUBLIC JACK_WAVE1_CONTINUOUS_SILENCE=1)
add_executable(picoOpenOmfNative "platform/native/board_omf.cpp" "platform/native/board_files.c" "platform/native/board_posix.c")
target_sources(picoOpenOmfNative PRIVATE "platform/native/native_game_menu.cpp")

target_link_libraries(picoOpenOmfNative PRIVATE omf_core jack_wave1_runtime hardware_psram)
target_compile_definitions(picoOpenOmfNative PRIVATE PICO_STACK_SIZE=32768 PICO_CORE1_STACK_SIZE=4096 PICO_USE_STACK_GUARDS=1
  NATIVE_GAME_DIRECTORY="/roms/OPENOMF")
pico_add_linker_script_override_path(picoOpenOmfNative "${PORT_ROOT}/cmake/linker")
target_link_options(picoOpenOmfNative PRIVATE -Wl,--gc-sections -Wl,--print-memory-usage)
pico_enable_stdio_uart(picoOpenOmfNative 1)
pico_enable_stdio_usb(picoOpenOmfNative 0)
pico_set_program_name(picoOpenOmfNative "picoOpenOmfNative")
option(OMF_TIMING_DIAG "Enable OpenOMF timing diagnostics" ON)
if(OMF_TIMING_DIAG)
  target_compile_definitions(omf_core PUBLIC OMF_TIMING_DIAG=1)
  target_sources(omf_core PRIVATE "platform/native/omf_timing.c")
  option(OMF_XSTEP "Precompute horizontal source coordinates" ON)
  if(OMF_XSTEP)
    target_compile_definitions(omf_core PRIVATE OMF_XSTEP=1)
    option(OMF_FAST_VIDEO "Use half-resolution video with a cached scene background" ON)
    if(OMF_FAST_VIDEO)
      target_compile_definitions(omf_core PRIVATE OMF_FAST_VIDEO=1)
      set(OMF_RGB555_DIRECT ON)
      if(OMF_RGB555_DIRECT)
        target_compile_definitions(omf_core PUBLIC OMF_RGB555_DIRECT=1)
        option(OMF_PALETTE_FAST "Fast palette lookup for unmodified pixels" ON)
        if(OMF_PALETTE_FAST)
          target_compile_definitions(omf_core PRIVATE OMF_PALETTE_FAST=1)
          option(OMF_SCENE_TIMING "Split scene draw timings" ON)
          if(OMF_SCENE_TIMING)
            target_compile_definitions(omf_core PRIVATE OMF_SCENE_TIMING=1)
            option(OMF_ROW_DIRTY "Restore dirty background ranges per row" ON)
            if(OMF_ROW_DIRTY)
              target_compile_definitions(omf_core PRIVATE OMF_ROW_DIRTY=1)
              option(OMF_SRAM_WORK "Place active half-resolution pixels in SRAM" ON)
              if(OMF_SRAM_WORK)
                target_compile_definitions(omf_core PRIVATE OMF_SRAM_WORK=1)
                option(OMF_SRAM_ARGB "Place active half-resolution ARGB in SRAM" ON)
                if(OMF_SRAM_ARGB)
                  target_compile_definitions(omf_core PRIVATE OMF_SRAM_ARGB=1)
                  option(OMF_ASYNC_LOG "Buffer UART diagnostic output without waiting for TX" ON)
                  if(OMF_ASYNC_LOG)
                    target_compile_definitions(picoOpenOmfNative PRIVATE OMF_ASYNC_LOG=1)
                    target_sources(picoOpenOmfNative PRIVATE "platform/native/omf_log_async.c")
                    option(OMF_LOOP_DIAG "Profile main-loop phases and PCM gap attribution" ON)
                    if(OMF_LOOP_DIAG)
                      target_compile_definitions(omf_core PUBLIC OMF_LOOP_DIAG=1)
                      target_sources(omf_core PRIVATE "platform/native/omf_loop_diag.c")
                      option(OMF_DYNAMIC_DIAG "Split dynamic tick timing into subphases" ON)
                      if(OMF_DYNAMIC_DIAG)
                        target_compile_definitions(omf_core PUBLIC OMF_DYNAMIC_DIAG=1)
                        if(OMF_FILE_DIAG)
                          pico_set_program_version(picoOpenOmfNative "0.1.2-public")
                        elseif(OMF_SURFACE_NOZERO)
                          pico_set_program_version(picoOpenOmfNative "0.1.2-dev03-surfnozero")
                        elseif(OMF_LOAD_DIAG)
                          pico_set_program_version(picoOpenOmfNative "0.1.2-dev03-loaddiag")
                        elseif(OMF_DECODE_COPY)
                          pico_set_program_version(picoOpenOmfNative "0.1.2-dev03-decodecopy-fault2")
                        elseif(OMF_CAPTURE_INDEX)
                          pico_set_program_version(picoOpenOmfNative "0.1.2-dev03-capindex")
                        elseif(OMF_CAPTURE_ROW)
                          pico_set_program_version(picoOpenOmfNative "0.1.2-dev03-caprow")
                        elseif(OMF_FAULT_DIAG)
                          pico_set_program_version(picoOpenOmfNative "0.1.2-dev03-faultdiag")
                        elseif(OMF_SPRITE_AUDIO)
                          pico_set_program_version(picoOpenOmfNative "0.1.2-dev03-spriteaudio")
                        elseif(OMF_READ_AUDIO)
                          pico_set_program_version(picoOpenOmfNative "0.1.2-dev03-readaudio")
                        else()
                          pico_set_program_version(picoOpenOmfNative "0.1.2-dev03-caproi")
                        endif()
                      else()
                        pico_set_program_version(picoOpenOmfNative "0.1.2-dev03-loopdiag")
                      endif()
                    else()
                      pico_set_program_version(picoOpenOmfNative "0.1.2-dev03-asynclog")
                    endif()
                  else()
                    pico_set_program_version(picoOpenOmfNative "0.1.2-dev03-sramargb")
                  endif()
                else()
                  pico_set_program_version(picoOpenOmfNative "0.1.2-dev03-sramwork")
                endif()
              else()
                pico_set_program_version(picoOpenOmfNative "0.1.2-dev03-rowdirty")
              endif()
            else()
              pico_set_program_version(picoOpenOmfNative "0.1.2-dev03-scene-diag")
            endif()
          else()
            pico_set_program_version(picoOpenOmfNative "0.1.2-dev03-palfast")
          endif()
        else()
          pico_set_program_version(picoOpenOmfNative "0.1.2-dev03-rgb555")
        endif()
      else()
        pico_set_program_version(picoOpenOmfNative "0.1.2-dev03-fastvideo")
      endif()
    else()
      pico_set_program_version(picoOpenOmfNative "0.1.2-dev03-xstep")
    endif()
  else()
    pico_set_program_version(picoOpenOmfNative "0.1.2-dev03-timing2")
  endif()
else()
  pico_set_program_version(picoOpenOmfNative "0.1.2-native-dev03")
endif()
include("${PORT_ROOT}/cmake/LoaderPartition.cmake")
omf_link_for_loader(picoOpenOmfNative)
pico_add_extra_outputs(picoOpenOmfNative)

  if(NOT OMF_LOOP_DIAG)
    message(FATAL_ERROR "OMF_FILE_DIAG requires OMF_LOOP_DIAG")
  endif()
  target_compile_definitions(omf_core PUBLIC OMF_FILE_DIAG=1)

if(OMF_READ_AUDIO)
  target_compile_definitions(picoOpenOmfNative PRIVATE OMF_READ_AUDIO=1)

if(OMF_SURFACE_NOZERO)
  target_compile_definitions(omf_core PRIVATE OMF_SURFACE_NOZERO=1)
endif()
if(OMF_LOAD_DIAG)
  if(NOT OMF_SPRITE_AUDIO)
    message(FATAL_ERROR "OMF_LOAD_DIAG requires OMF_SPRITE_AUDIO")

  target_compile_definitions(omf_core PRIVATE OMF_LOAD_DIAG=1)

  target_compile_definitions(omf_core PRIVATE OMF_SPRITE_AUDIO=1)

if(OMF_FAULT_DIAG)
  target_compile_definitions(omf_core PUBLIC OMF_FAULT_DIAG=1)
  target_sources(omf_core PRIVATE "platform/native/omf_fault_diag.c")
  target_link_libraries(omf_core PRIVATE hardware_exception)

option(OMF_CAPTURE_ROW "Use SRAM row staging for full-resolution capture raster" OFF)
if(OMF_CAPTURE_ROW)
  if(NOT OMF_FAST_VIDEO)
    message(FATAL_ERROR "OMF_CAPTURE_ROW requires OMF_FAST_VIDEO")
  endif()
  target_compile_definitions(omf_core PRIVATE OMF_CAPTURE_ROW=1)
endif()

if(OMF_CAPTURE_INDEX)

    message(FATAL_ERROR "OMF_CAPTURE_INDEX needs SRAM_WORK/FAST_VIDEO and excludes CAPTURE_ROW")
  endif()
  target_compile_definitions(omf_core PRIVATE OMF_CAPTURE_INDEX=1)
endif()

if(OMF_DECODE_COPY)
  target_compile_definitions(omf_core PRIVATE OMF_DECODE_COPY=1)
